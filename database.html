async function runAIAnalysis() { const name = document.getElementById("ai-species").value.trim(); if (!name) return alert("请输入植物名称"); const box = document.getElementById("ai-result"); box.style.display = "block"; box.innerHTML = "🧠 AI 正在分析..."; try { const res = await fetch(`${SB_URL}/functions/v1/analyze-species`, { method: "POST", headers: { "Content-Type": "application/json", "apikey": SB_KEY, "Authorization": `Bearer ${SB_KEY}` }, body: JSON.stringify({ name: name }) }); const d = await res.json(); if (!d.success) throw new Error(d.error || "未知错误"); box.innerHTML = `

🔬 ${name} 分析报告

${d.result}
`; } catch (e) { box.innerHTML = `❌ 错误：${e.message}`; } }
