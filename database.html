function loadSpecies() { var tbody = document.getElementById("sp-body"); get("/rest/v1/species?select=*", false) .then(function(d) { if (!d || !d.length) { tbody.innerHTML = '暂无物种数据'; return; } document.getElementById("cnt-s").textContent = d.length; tbody.innerHTML = d.map(function(r, i) { return ` ${pad(i + 1)} ${r.name_latin || "—"} ${r.name_zh || "—"} ${r.description || "—"} `; }).join(""); }) .catch(function(e) { tbody.innerHTML = '❌ ' + e.message + ''; }); }
